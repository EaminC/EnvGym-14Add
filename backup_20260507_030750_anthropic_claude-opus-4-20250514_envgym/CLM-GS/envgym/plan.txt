     # Get system memory info
     memory = psutil.virtual_memory()
     print(f"Total RAM: {memory.total / (1024**3):.2f} GB")
     print(f"Available RAM: {memory.available / (1024**3):.2f} GB")
     print(f"Used RAM: {memory.used / (1024**3):.2f} GB ({memory.percent}%)")
     
     # Test large tensor allocation
     try:
         size_gb = 1
         elements = int(size_gb * 1024**3 / 4)  # float32 = 4 bytes
         tensor = torch.randn(elements)
         print(f"\n✓ Successfully allocated {size_gb} GB tensor")
         del tensor
     except Exception as e:
         print(f"\n✗ Failed to allocate {size_gb} GB tensor: {e}")
